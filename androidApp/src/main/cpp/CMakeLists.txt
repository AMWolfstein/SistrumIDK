cmake_minimum_required(VERSION 3.22.1)
project(sistrum_player LANGUAGES C CXX)

set(FFMPEG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}")
set(FFMPEG_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/ffmpeg/include")

if(NOT EXISTS "${FFMPEG_INCLUDE}/libavcodec/avcodec.h")
    message(FATAL_ERROR "FFmpeg artifacts are missing. Run: bash scripts/build-ffmpeg-android.sh ${ANDROID_ABI}")
endif()

option(SISTRUM_WITH_TAGLIB "Link the TagLib metadata bridge" ON)
if(SISTRUM_WITH_TAGLIB)
    set(TAGLIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/taglib/${ANDROID_ABI}")
    if(NOT EXISTS "${TAGLIB_ROOT}/lib/libtag.a")
        message(FATAL_ERROR "TagLib artifacts are missing. Run: bash scripts/build-taglib-android.sh ${ANDROID_ABI}")
    endif()
    add_library(tag STATIC IMPORTED)
    set_target_properties(tag PROPERTIES IMPORTED_LOCATION "${TAGLIB_ROOT}/lib/libtag.a")
endif()

foreach(library avutil swresample avcodec avformat avfilter)
    add_library(${library} SHARED IMPORTED)
    set_target_properties(${library} PROPERTIES IMPORTED_LOCATION "${FFMPEG_ROOT}/lib${library}.so")
endforeach()

add_library(sistrum_player SHARED ffmpeg_player.cpp $<$<BOOL:${SISTRUM_WITH_TAGLIB}>:taglib_bridge.cpp>)
if(SISTRUM_WITH_TAGLIB)
    target_include_directories(sistrum_player PRIVATE "${TAGLIB_ROOT}/include")
endif()
target_include_directories(sistrum_player PRIVATE "${FFMPEG_INCLUDE}")
target_compile_options(sistrum_player PRIVATE -Oz -ffunction-sections -fdata-sections)
target_link_options(sistrum_player PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
if(SISTRUM_WITH_TAGLIB)
    target_link_libraries(sistrum_player tag)
endif()
target_link_libraries(sistrum_player avformat avcodec avfilter swresample avutil aaudio android log)
